```ts
export interface Cell { x: number; y: number }
function snake(): Cell[] {
  const cells: Cell[] = [];
  const rows = [1, 3, 5, 7];
  let dir = 1;
  cells.push({ x: 0, y: 1 });
  for (const y of rows) {
    for (let x = dir > 0 ? 1 : 7; dir > 0 ? x <= 8 : x >= 0; x += dir) cells.push({ x, y });
    if (y < 7) cells.push({ x: dir > 0 ? 8 : 0, y: y + 1 }, { x: dir > 0 ? 8 : 0, y: y + 2 });
    dir = -dir;
  }
  cells.push({ x: 8, y: 8 });
  return cells;
}
export const PATH = snake();
export interface WaveDef { count: number; hpMul: number; speed: number; tankEvery: number; runnerEvery: number }
export const WAVES: WaveDef[] = Array.from({ length: 10 }, (_, i) => ({
  count: 10 + Math.round((40 - 10) * (i / 9)),
  hpMul: 1.35 ** i,
  speed: 1 + i * 0.06,
  tankEvery: i < 2 ? 0 : 4,
  runnerEvery: i < 3 ? 0 : 3,
}));
export interface GameState { wave: number; lives: number; coins: number; units: Unit[]; turrets: Turret[]; buffs: string[]; over: boolean; stars: number; dmgMul?: number; rateMul?: number; pierce?: boolean; pugs?: number; saleMul?: number }
export function createGame(): GameState {
  return { wave: 0, lives: 10, coins: 200, units: [], turrets: [], buffs: [], over: false, stars: 0 };
}
// Ruling контроллера: казна 200 (не 120) — иначе scarlet за 150 не купить в verbatim-тесте линии T2; спека стартовую казну не задаёт, тест T1 казну не проверяет.
export interface EnemyDef { hp: number; speed: number; reward: number }
export const ENEMIES: Record<string, EnemyDef> = {
  zevaka: { hp: 20, speed: 1.0, reward: 6 },
  zanuda: { hp: 60, speed: 0.6, reward: 12 },
  sprinter: { hp: 12, speed: 1.7, reward: 8 },
  director: { hp: 400, speed: 0.5, reward: 200 },
};
```

```ts
export interface TurretDef { cost: number; dmg: number; rate: number; range: number }
export const TURRETS: Record<string, TurretDef> = {
  flood: { cost: 30, dmg: 9, rate: 2.5, range: 3.0 },
  cobalt: { cost: 70, dmg: 30, rate: 1.4, range: 3.6 },
  scarlet: { cost: 150, dmg: 60, rate: 0.9, range: 4.0 },
};
// Финальные числа после сим-прогона T4 (было 4/2/2.2, 12/1/2.8, 30/0.6/3.4); цены не менялись; спека-числа не тронуты.
export function placeTurret(g: GameState, x: number, y: number, kind: string): boolean {
  const t = TURRETS[kind];
  if (!t || g.coins < t.cost) return false;
  if (PATH.some((c) => c.x === x && c.y === y)) return false;
  if (g.turrets.some((u) => u.x === x && u.y === y)) return false;
  if (g.turrets.length >= 12) return false;
  g.coins -= t.cost;
  g.turrets.push({ x, y, kind, cd: 0 });
  return true;
}
export function sellTurret(g: GameState, i: number): void {
  const [t] = g.turrets.splice(i, 1);
  if (t) g.coins += Math.floor(TURRETS[t.kind].cost * 0.7);
}
```

(`tick`/`spawnWave`/`Unit`: юнит `{kind, seg, pos, hp, maxHp, speed, reward}`; `tick` двигает `pos += speed*0.1`, на конце PATH — `lives--`, турели бьют ближайшего в `range` по `rate`; scarlet бьёт по линии — урон цели и всем юнитам в 1 клетке от линии выстрела.)

```ts
export interface Unit { kind: string; seg: number; pos: number; hp: number; maxHp: number; speed: number; reward: number }
export interface Turret { x: number; y: number; kind: string; cd: number }
export function spawnWave(g: GameState, w: number): void {
  const def = WAVES[w];
  for (let i = 0; i < def.count; i++) {
    let kind = 'zevaka';
    if (def.tankEvery > 0 && i % def.tankEvery === def.tankEvery - 1) kind = 'zanuda';
    if (def.runnerEvery > 0 && i % def.runnerEvery === def.runnerEvery - 1) kind = 'sprinter';
    const e = ENEMIES[kind];
    g.units.push({ kind, seg: 0, pos: -i * 0.5, hp: e.hp * def.hpMul, maxHp: e.hp * def.hpMul, speed: e.speed * def.speed, reward: e.reward });
  }
  if (w === 9) {
    const e = ENEMIES.director;
    g.units.push({ kind: 'director', seg: 0, pos: -def.count * 0.5 - 2, hp: e.hp * def.hpMul, maxHp: e.hp * def.hpMul, speed: e.speed * def.speed, reward: e.reward });
  }
}
function cellOf(u: Unit): { x: number; y: number } {
  const a = PATH[Math.max(0, Math.min(u.seg, PATH.length - 1))];
  const b = PATH[Math.min(u.seg + 1, PATH.length - 1)];
  return { x: a.x + (b.x - a.x) * u.pos, y: a.y + (b.y - a.y) * u.pos };
}
// Ruling контроллера: убитые остаются трупами (dead=true, монета один раз) — требует тест линии; дошедшие до штаба удаляются из массива. Живой юнит = !dead. Task 4: проверки «волна чиста» — по alive-счёту, не по units.length. Task 5: вьюха трупы не рисует.
// В tick для scarlet: цель — ближайший в range; урон получают цель и все юниты,
// чьё расстояние до отрезка «турель→цель» ≤ 1 клетка (линия; с картой pierce — до 2 клеток).
```

```ts
export const CARDS: Record<string, { name: string; desc: string }> = {
  rate: { name: 'Двойной прожектор', desc: 'скорострельность +30%' },
  dmg: { name: 'Кобальтовый кулак', desc: 'урон +40%' },
  pierce: { name: 'Алый коридор', desc: 'лазер бьёт насквозь' },
  repair: { name: 'Ремонт штаба', desc: '+3 жизни' },
  pugs: { name: 'Мопс-подкрепление', desc: 'кусает ближайшего, 5 волн' },
  sale: { name: 'Скидка прапора', desc: 'турели −25%' },
};
export function applyCard(g: GameState, id: string): void {
  const n = g.buffs.filter((b) => b === id).length;
  const k = 0.7 ** n;
  g.buffs.push(id);
  if (id === 'rate') g.rateMul = (g.rateMul || 1) + 0.3 * k;
  if (id === 'dmg') g.dmgMul = (g.dmgMul || 1) + 0.4 * k;
  if (id === 'repair') g.lives = Math.min(10, g.lives + 3);
// Ruling контроллера: стак аддитивный (+0.4k), не мультипликативный — скетч давал 1.792, а verbatim-тест требует 1.68; тест диктует.
}
export function finishWave(g: GameState, lost: number): void {
  g.stars = lost === 0 ? 3 : lost <= 3 ? 2 : 1;
}
```

```ts
import { describe, expect, test } from 'bun:test';
import { createGame, placeTurret, applyCard, tick, spawnWave } from './engine';
function bot(): { won: boolean; stars: number } {
  const g = createGame();
  const spots = [[4,0],[4,2],[2,4],[6,4],[1,6],[7,6],[3,8],[5,8]];
  let si = 0;
  for (let w = 0; w < 10; w++) {
    while (si < spots.length && placeTurret(g, spots[si][0], spots[si][1], si % 3 === 2 ? 'cobalt' : 'flood')) si++;
    spawnWave(g, w);
    for (let t = 0; t < 3000 && g.units.length > 0 && g.lives > 0; t++) tick(g);
    if (g.lives <= 0) return { won: false, stars: 0 };
    if (w < 9) applyCard(g, 'dmg');
  }
  return { won: g.lives > 0, stars: g.stars };
}
describe('сим-прогон', () => {
  test('бот с турелями и баффами проходит 10 волн', () => {
    expect(bot().won).toBe(true);
  });
  test('голый штаб без турелей падает', () => {
    const g = createGame();
    for (let w = 0; w < 10; w++) {
      spawnWave(g, w);
      for (let t = 0; t < 3000 && g.units.length > 0 && g.lives > 0; t++) tick(g);
      if (g.lives <= 0) break;
    }
    expect(g.lives <= 0).toBe(true);
  });
});
```
